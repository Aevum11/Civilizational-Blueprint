        if self.r1 <= 0 or self.r2 <= 0: return float('inf')

    @property
    def lattice_distance(self) -> int:

        k = self.lattice_distance

        return S // g

    def current_variance(self) -> float:

        return (LOCKED - SUBLIMINAL) / V_BASE  # = 5
